
# ==============================================================================

# Designed for the "EF5 Dynamic Dockerization" workflow.

# they appear in the manual.
# ==============================================================================

# ==============================================================================
# BASIC BLOCK
# ==============================================================================
# Defines the core domain and geographic projection details.
[Basic MySimu]
DEM=/data/basic/your_DEM.tif
DDM=/data/basic/your_DDM.tif
FAM=/data/basic/your_FAM.tif
PROJ=geographic
ESRIDDM=true
SELFFAM=false

# ==============================================================================
# PRECIP FORCING BLOCK
# ==============================================================================
# Defines the precipitation inputs.
# In the Docker container, ensure these paths point to the read-write /data
# directory mapped from your host machine.
[PrecipForcing MyPrecip]
Type=BIF
UNIT=mm/h
FREQ=10u
LOC=/data/precip/
NAME=Q2_YYYYMMDDUU.bif

# ==============================================================================
# TEMP FORCING BLOCK
# ==============================================================================

# Strictly required if you are using the Snow17 model to simulate snow accumulation/melt.

#Type=TIF
#UNIT=C
#FREQ=1h
#LOC=/data/temp/
#NAME=temp_YYYYMMDDHH.tif

# ==============================================================================
# PET FORCING BLOCK
# ==============================================================================
# Defines the Potential Evapotranspiration (PET) inputs.
# These paths must also point to the /data mount within the container.
[PETForcing PETclimo]
LOC=/data/pet/
FREQ=1mo
UNIT=mm/d
NAME=PET_MM.tif
TYPE=TIF

# ==============================================================================
# GAUGE BLOCK
# ==============================================================================
# Defines observation gauges for flow comparison or calibration. A gauge block is made for each 
# individual gauge you wish to model. Alternatively to using lat/lon, you can use the grid definition
# locations celly/cellx
[Gauge USGS-08104900]
Lat=30.62575
Lon=-97.69112
obs=/data/observations/usgs/USGS-08104900.csv
#basinarea=344          #drainage area in km^2
#outputts=False         #output a time series of discharge
#wantda=False           #if we are doing data assimilation are we using this gauge?
#wantco=False           #will this gauge be included in a combined output file?

# You can also lay out your gauge blocks horizontally to save space for dense gauge networks! Example:
[Gauge USGS-08104900] Lat=30.62575 Lon=-97.69112 obs=/data/observations/usgs/USGS-08104900.csv
